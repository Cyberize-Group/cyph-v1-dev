4: /**
5:  * Fetches the user's role from the user_roles table in the database.
6:  * This is the canonical source of truth for authorization.
7:  * 
8:  * @param userId - The authenticated user's ID from auth.users
9:  * @returns The user's role or null if not found
10:  */
11: export const getUserRole = async (userId: string): Promise<AppRole | null> => {

25:   }

67: CREATE OR REPLACE FUNCTION public.handle_new_user()
68: RETURNS trigger
69: LANGUAGE plpgsql
70: SECURITY DEFINER
71: SET search_path = public
72: AS $$
73: DECLARE
74:   assigned_role public.app_role;
75: BEGIN
76:   IF NEW.raw_user_meta_data ->> 'role' IS NOT NULL THEN
77:     assigned_role := (NEW.raw_user_meta_data ->> 'role')::public.app_role;
78:   ELSE
79:     assigned_role := 'member'::public.app_role;
80:   END IF;
81: 
82:   INSERT INTO public.user_roles (user_id, role)
83:   VALUES (NEW.id, assigned_role)
84:   ON CONFLICT (user_id) DO NOTHING;
85: 
86:   INSERT INTO public.profiles (id, email, full_name)
87:   VALUES (
88:     NEW.id,
89:     NEW.email,
90:     NEW.raw_user_meta_data ->> 'full_name'
91:   )
92:   ON CONFLICT (id) DO NOTHING;
93: 
94:   RETURN NEW;
95: END;
96: $$;

108: RETURNS event_trigger

110: AS $$
111: DECLARE
112:   obj record;
113: BEGIN
114:   FOR obj IN

116:   LOOP

118:       EXECUTE format('ALTER TABLE %s ENABLE ROW LEVEL SECURITY', obj.object_identity);
